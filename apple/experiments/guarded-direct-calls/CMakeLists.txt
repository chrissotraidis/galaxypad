cmake_minimum_required(VERSION 3.20)
project(galaxypad_guarded_direct_calls C)

# Diagnostic-only wrapper. The vendor target retains its normal flags, ABI
# tables and runtime sources; explicitly selected chunks are replaced in this
# build graph only. Neither generated source nor production CMake is edited.
get_filename_component(GALAXYPAD_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(GALAXYPAD_DIRECT_CHUNK "" CACHE FILEPATH "Original generated chunk to replace")
set(GALAXYPAD_DIRECT_OVERRIDE "" CACHE FILEPATH "Isolated transformed chunk")
set(GALAXYPAD_DIRECT_OVERLAY "" CACHE PATH "Completed multi-chunk diagnostic overlay")
if(NOT GALAXYPAD_DIRECT_OVERLAY)
foreach(INPUT GALAXYPAD_DIRECT_CHUNK GALAXYPAD_DIRECT_OVERRIDE)
  if(NOT EXISTS "${${INPUT}}")
    message(FATAL_ERROR "${INPUT} must name an existing source file")
  endif()
  file(REAL_PATH "${${INPUT}}" ${INPUT}_REAL)
endforeach()
if(GALAXYPAD_DIRECT_CHUNK_REAL STREQUAL GALAXYPAD_DIRECT_OVERRIDE_REAL)
  message(FATAL_ERROR "Override must be separate from original source")
endif()
endif()

add_subdirectory("${GALAXYPAD_ROOT}/ref/ModernGekko/vendor/dolphin/module-template"
                 "module")
set(EXPERIMENT_TARGET "g${GAME_ID}_recomp")
get_target_property(EXPERIMENT_SOURCES ${EXPERIMENT_TARGET} SOURCES)
if(GALAXYPAD_DIRECT_OVERLAY)
  file(READ "${GALAXYPAD_DIRECT_OVERLAY}/manifest.json" OVERLAY_MANIFEST)
  string(JSON OVERLAY_SOURCE GET "${OVERLAY_MANIFEST}" generated)
  file(REAL_PATH "${GENERATED_DIR}" SELECTED_SOURCE)
  if(NOT OVERLAY_SOURCE STREQUAL SELECTED_SOURCE)
    message(FATAL_ERROR "Overlay was prepared from another generated tree")
  endif()
  string(JSON OVERLAY_COUNT LENGTH "${OVERLAY_MANIFEST}" files)
  math(EXPR OVERLAY_LAST "${OVERLAY_COUNT}-1")
  foreach(INDEX RANGE ${OVERLAY_LAST})
    string(JSON RECORD GET "${OVERLAY_MANIFEST}" files ${INDEX})
    string(JSON ORIGINAL GET "${RECORD}" source)
    string(JSON EXPECTED GET "${RECORD}" source_sha256)
    file(SHA256 "${ORIGINAL}" ACTUAL)
    if(NOT ACTUAL STREQUAL EXPECTED)
      message(FATAL_ERROR "Overlay source changed: ${ORIGINAL}")
    endif()
    string(JSON SITES GET "${RECORD}" sites)
    if(SITES GREATER 0)
      string(JSON OVERRIDE GET "${RECORD}" override)
      string(JSON EXPECTED GET "${RECORD}" override_sha256)
      file(SHA256 "${OVERRIDE}" ACTUAL)
      if(NOT ACTUAL STREQUAL EXPECTED)
        message(FATAL_ERROR "Overlay candidate changed: ${OVERRIDE}")
      endif()
      list(FIND EXPERIMENT_SOURCES "${ORIGINAL}" CHUNK_INDEX)
      if(CHUNK_INDEX LESS 0)
        message(FATAL_ERROR "Overlay source not in selected target: ${ORIGINAL}")
      endif()
      list(REMOVE_AT EXPERIMENT_SOURCES ${CHUNK_INDEX})
      list(APPEND EXPERIMENT_SOURCES "${OVERRIDE}")
    endif()
  endforeach()
  message(STATUS "Validated diagnostic overlay ${GALAXYPAD_DIRECT_OVERLAY}")
else()
list(FIND EXPERIMENT_SOURCES "${GALAXYPAD_DIRECT_CHUNK_REAL}" CHUNK_INDEX)
if(CHUNK_INDEX LESS 0)
  message(FATAL_ERROR "Original chunk is not in the selected module target")
endif()
list(REMOVE_AT EXPERIMENT_SOURCES ${CHUNK_INDEX})
list(APPEND EXPERIMENT_SOURCES "${GALAXYPAD_DIRECT_OVERRIDE_REAL}")
endif()
list(APPEND EXPERIMENT_SOURCES "${CMAKE_CURRENT_LIST_DIR}/binding.c")
set_property(TARGET ${EXPERIMENT_TARGET} PROPERTY SOURCES "${EXPERIMENT_SOURCES}")
message(STATUS "Guarded diagnostic configured; no installed module selection changes")
